   336→function ImageGallery({ currentVariantImage }: { currentVariantImage: string }) {
   337→  const [currentImage, setCurrentImage] = useState(0)
   338→  const [isAutoPlaying, setIsAutoPlaying] = useState(true)
   339→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null)
   340→  const scrollRef = useRef<HTMLDivElement>(null)
   341→
   342→  const images = useMemo(() => {
   343→    const filtered = productImages.filter((img) => img !== currentVariantImage)
   344→    return [currentVariantImage, ...filtered]
   345→  }, [currentVariantImage])
   346→
   347→  const goTo = useCallback(
   348→    (index: number) => {
   349→      setCurrentImage((index + images.length) % images.length)
   350→    },
   351→    [images.length]
   352→  )
   353→
   354→  const next = useCallback(() => goTo(currentImage + 1), [currentImage, goTo])
   355→  const prev = useCallback(() => goTo(currentImage - 1), [currentImage, goTo])
   356→
   357→  // Auto-play
   358→  useEffect(() => {
   359→    if (isAutoPlaying) {
   360→      intervalRef.current = setInterval(() => {
   361→        setCurrentImage((prev) => (prev + 1) % images.length)
   362→      }, 4000)
   363→    }
   364→    return () => {
   365→      if (intervalRef.current) clearInterval(intervalRef.current)
   366→    }
   367→  }, [isAutoPlaying, images.length])
   368→
   369→  // Scroll thumbnail into view
   370→  useEffect(() => {
   371→    if (scrollRef.current) {
   372→      const thumb = scrollRef.current.children[currentImage] as HTMLElement | undefined
   373→      if (thumb) {
   374→        thumb.scrollIntoView({ behavior: 'smooth', inline: 'center', block: 'nearest' })
   375→      }
   376→    }
   377→  }, [currentImage])
   378→
   379→  return (
   380→    <div className="relative">
   381→      {/* Main image */}
   382→      <div
   383→        className="relative aspect-square bg-stone-100 overflow-hidden rounded-xl"
   384→        onTouchStart={() => setIsAutoPlaying(false)}
   385→        onMouseDown={() => setIsAutoPlaying(false)}
   386→      >
   387→        <img
   388→          src={images[currentImage]}
   389→          alt={`Product image ${currentImage + 1}`}
   390→          className="w-full h-full object-cover transition-opacity duration-300"
   391→        />
   392→
   393→        {/* Navigation arrows */}
   394→        <button
   395→          onClick={prev}
   396→          className="absolute left-2 top-1/2 -translate-y-1/2 bg-white rounded-full shadow-md p-2 hover:bg-stone-50 transition-colors"
   397→          aria-label="Previous image"
   398→        >
   399→          <ChevronLeft className="h-4 w-4 text-zinc-700" />
   400→        </button>
   401→        <button
   402→          onClick={next}
   403→          className="absolute right-2 top-1/2 -translate-y-1/2 bg-white rounded-full shadow-md p-2 hover:bg-stone-50 transition-colors"
   404→          aria-label="Next image"
   405→        >
   406→          <ChevronRight className="h-4 w-4 text-zinc-700" />
   407→        </button>
   408→
   409→        {/* Dot indicators */}
   410→        <div className="absolute bottom-3 left-1/2 -translate-x-1/2 flex items-center gap-1.5">
   411→          {images.slice(0, 8).map((_, idx) => (
   412→            <div
   413→              key={idx}
   414→              className={`rounded-full transition-all duration-300 ${
   415→                idx === currentImage
   416→                  ? 'bg-white w-5 h-2'
   417→                  : 'bg-white/50 w-2 h-2'
   418→              }`}
   419→            />
   420→          ))}
   421→        </div>
   422→      </div>
   423→
   424→      {/* Thumbnail strip */}
   425→      <div
   426→        ref={scrollRef}
   427→        className="flex gap-2 mt-3 overflow-x-auto scrollbar-hide px-1 pb-1"
   428→        style={{ scrollbarWidth: 'none', msOverflowStyle: 'none' }}
   429→      >
   430→        {images.slice(0, 8).map((img, idx) => (
   431→          <button
   432→            key={img}
   433→            onClick={() => {
   434→              setCurrentImage(idx)
   435→              setIsAutoPlaying(false)
   436→            }}
   437→            className={`flex-shrink-0 h-16 w-16 rounded-lg overflow-hidden border-2 transition-all duration-200 ${
   438→              idx === currentImage
   439→                ? 'border-zinc-900 opacity-100'
   440→                : 'border-transparent opacity-50 hover:opacity-75'
   441→            }`}
   442→          >
   443→            <img src={img} alt={`Thumbnail ${idx + 1}`} className="w-full h-full object-cover" />
   444→          </button>
   445→        ))}
   446→      </div>
   447→    </div>
   448→  )
   449→}

   453→function Accessories() {

   537→        setError(data.error || t('tracking.notFound'))
   538→        return
   539→      }
   540→
   541→      setResult(data)
   542→    } catch {
   543→      setError(t('tracking.searchError'))

   545→      setLoading(false)
   546→    }
